<?php

class AuthController extends BaseController

    public function register(Request $request)
    {
        $validator = Validator::make($request->all(), [
            'name' => 'required|string|max:255',
            'email' => 'required|string|email|max:255|unique:users',
            'password' => 'required|string|min:8',
        ]);

        if ($validator->fails()) {
            return response()->json($validator->errors(), 422);
        }

        $user = User::create([
            'name' => $request->name,
            'email' => $request->email,
            'password' => Hash::make($request->password),
        ]);
        $role= Role::where('name', 'Viewer')->first();
        $user->roles()->attach($role->id);
        $tokenName = 'adminAuthToken';
        Auth::login($user);
        $permissions = $role->permissions()->pluck('name'); 
        return $this->success('User registered successfully',[
            'user' => $user,
            'token' => $user->createToken($tokenName)->plainTextToken,
            'roles'=>$role->name,
            'permissions'=>$permissions
        ]);
    }

    public function login(Request $request)
    {
       
        try {
            $rules = [
                'email' => 'required|string',
                'password' => 'required|string',
            ];
            $validator = Validator::make($request->all(), $rules);
            try{
                if ($validator->fails()) {
                    return $this->error($validator->errors()->first());
                } elseif (! Auth::attempt($request->only('email', 'password'))) {
                    
                    return $this->error('Credentials not match', 401);
                }
            }
            catch(\Exception $e){
                return $this->error($e->getMessage() );
            }
           
           
            // if (! Auth::user()->is_active) {
            //     return $this->error('Account is not Active!');
            // }
            // if (Auth::user()->hasRole(['owner', 'admin', 'super_admin'])) {
            $tokenName = 'adminAuthToken';
            $user = Auth::user();
            $role = $user->roles()->first();
           $permissions = $role->permissions()->pluck('name');

            return $this->success('Login Success', [
                'access_token' => $user->createToken($tokenName)->plainTextToken,
                'token_type' => 'Bearer',
                'user' => $user,
                'role' => $role->name,
                'permissions' => $permissions,
                
            ]);
            // }
        } catch (\Exception $exception) {
            return $this->error('Login Failed');
        }
    }

    
    public function user()
    {
        $user = Auth::user();
        $role = $user->roles()->with('permissions')->get();
        $role_names = $role->pluck('name')->implode(',');
        $permissions = $role->pluck('permissions')->flatten()->pluck('name');
        
        // Fix: Get token from request header for Sanctum
        $token = request()->bearerToken();
            
        return response()->json([
            'user' => $user,
            'token' => $token,
            'roles' => $role_names,
            'permissions' => $permissions,
        ]);
    }

    public function logout()

        return response()->json(['message' => 'Successfully logged out']);

    public function debugToken(Request $request)

            'all_headers' => $request->headers->all(),

    }
